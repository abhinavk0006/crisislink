19test("Real local embeddings and PostgreSQL vector workflow", async (t) => {
20 const db = await getDb();

24 await check("Real pgvector extension and HNSW cosine index exist", async () => {
25 assert.equal((await db.query("SELECT extversion FROM pg_extension WHERE extname='vector'")).rows.length, 1);
26 const index = (await db.query<{ indexdef: string }>("SELECT indexdef FROM pg_indexes WHERE indexname='idx_emergency_embedding_hnsw'")).rows[0];
27 assert.match(index.indexdef, /hnsw.*vector_cosine_ops/);
28 });
29 await check("Pinned MiniLM generates normalized 384-dimensional embeddings offline", async () => {
30 const a = await embedText("A physician is treating an injured person");
31 assert.equal(a.length, 384);
32 assert.ok(Math.abs(Math.hypot(...a) - 1) < 0.001);
33 assert.deepEqual(a, await embedText("A physician is treating an injured person"));
34 });
35 let target = 0;
36 await check("Meaning search ranks medical incident without requiring matching query words", async () => {
37 const incident = await createIncident(coordinator, { type_id: 2, location_id: 2,
38 description: "A physician is treating an injured person who needs urgent hospital transport.", severity_level: "high", people_affected: 1 });
39 target = incident.emergency_id;
40 const result = await searchIncidents(coordinator, { query: "An ambulance is needed for a patient", limit: 3 });
41 assert.equal((result.rows[0] as any).emergency_id, target);
42 assert.equal(result.coverage.indexed, result.coverage.total);
43 evidence.semanticExample = result;
44 });
45 await check("Repeated search skips up-to-date embeddings and preserves distance ordering", async () => {
46 assert.equal(await indexIncidents(), 0);
47 const result = await searchIncidents(technician, { query: "A warehouse is burning", limit: 3 });
48 assert.equal((result.rows[0] as any).emergency_id, 101);
49 const distances = result.rows.map((r: any) => r.cosine_distance);
50 assert.deepEqual(distances, [...distances].sort((a, b) => a - b));

52 await check("Description edits invalidate old vectors and the next search reindexes", async () => {
53 await db.query("UPDATE emergencies SET description=$1 WHERE emergency_id=$2", ["Floodwaters surround a home and residents require a rescue boat.", target]);

55 assert.deepEqual(row, { embedding: null, embedding_model: null, embedded_at: null });
