5// Incremental derived-index maintenance by the trusted backend, not a user write API.
6// Inference is outside SQL transactions; the conditional update prevents stale writes.
7export async function indexIncidents() {
8 const db = await getDb();
9 const pending = (await db.query<{ emergency_id: number; description: string }>(
10 "SELECT emergency_id,description FROM emergencies WHERE embedding IS NULL OR embedding_model IS DISTINCT FROM $1 ORDER BY emergency_id",
11 [EMBEDDING_ID],
12 )).rows;
13 let indexed = 0;
14 for (const incident of pending) {
15 const embedding = vectorLiteral(await embedText(incident.description));
16 const changed = await db.query(
17 "UPDATE emergencies SET embedding=$1::vector,embedding_model=$2,embedded_at=CURRENT_TIMESTAMP WHERE emergency_id=$3 AND description=$4 RETURNING emergency_id",
18 [embedding, EMBEDDING_ID, incident.emergency_id, incident.description],
19 );
20 indexed += changed.rows.length;
21 }
22 return indexed;
23}

25export const SEMANTIC_SQL = `SELECT emergency_id,description,status,severity_level,
26 embedding <=> $1::vector AS cosine_distance,
27 1 - (embedding <=> $1::vector) AS cosine_similarity
28FROM emergencies
29WHERE embedding IS NOT NULL AND embedding_model=$2
30ORDER BY embedding <=> $1::vector
31LIMIT $3`;

33export async function searchIncidents(actor: Actor, input: unknown) {
34 const { query, limit } = z.object({
35 query: z.string().trim().min(3).max(500),
36 limit: z.coerce.number().int().min(1).max(10).default(5),
37 }).parse(input);
38 const newlyIndexed = await indexIncidents();
39 const vector = vectorLiteral(await embedText(query));
40 return asActor(actor, async (tx) => {
41 const rows = (await tx.query(SEMANTIC_SQL, [vector, EMBEDDING_ID, limit])).rows;
42 const coverage = (await tx.query<{ total: number; indexed: number }>(
43 "SELECT COUNT(*)::int AS total,COUNT(*) FILTER(WHERE embedding IS NOT NULL AND embedding_model=$1)::int AS indexed FROM emergencies",
44 [EMBEDDING_ID],
45 )).rows[0];
46 return { rows, coverage, newlyIndexed, model: EMBEDDING_ID,
47 dimensions: EMBEDDING_DIMENSIONS, metric: "cosine distance", sql: SEMANTIC_SQL };
48 });
