4// Incremental derived-index maintenance by the trusted backend, not a user write API.
5// Inference is outside SQL transactions; the conditional update prevents stale writes.
6export async function indexIncidents() {
7 const db = await getDb();
8 const pending = (await db.query<{ emergency_id: number; description: string }>(
9 "SELECT emergency_id,description FROM emergencies WHERE embedding IS NULL OR embedding_model IS DISTINCT FROM $1 ORDER BY emergency_id",
10 [EMBEDDING_ID],
11 )).rows;
12 let indexed = 0;
13 for (const incident of pending) {
14 const embedding = vectorLiteral(await embedText(incident.description));
15 const changed = await db.query(
16 "UPDATE emergencies SET embedding=$1::vector,embedding_model=$2,embedded_at=CURRENT_TIMESTAMP WHERE emergency_id=$3 AND description=$4 RETURNING emergency_id",
17 [embedding, EMBEDDING_ID, incident.emergency_id, incident.description],
18 );
19 indexed += changed.rows.length;
20 }
21 return indexed;
22}
