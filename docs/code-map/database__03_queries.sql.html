1-- JOIN: show incident type and location.
2SELECT e.emergency_id,t.type_name,l.location_name,e.status
3FROM emergencies e JOIN emergency_types t USING(type_id)
4JOIN locations l USING(location_id) ORDER BY e.emergency_id;
5
6-- Nested scalar query: incidents above the overall affected-people average.
7SELECT emergency_id,people_affected FROM emergencies
8WHERE people_affected>(SELECT AVG(people_affected) FROM emergencies);
9
10-- Correlated query: incidents with no active assignment.
11SELECT e.emergency_id,e.status FROM emergencies e WHERE NOT EXISTS(
12 SELECT 1 FROM assignments a WHERE a.emergency_id=e.emergency_id
13 AND a.status IN ('active','dispatched'));
14
15-- CTE and window function: latest audit event per incident.
16WITH ranked AS (
17 SELECT history_id,emergency_id,status,changed_at,
18 ROW_NUMBER() OVER(PARTITION BY emergency_id ORDER BY changed_at DESC,history_id DESC) AS rn
19 FROM emergency_status_history
20) SELECT * FROM ranked WHERE rn=1;
21
22-- Recursive CTE: seven-day reporting calendar, including dates with zero incidents.
23WITH RECURSIVE days(day) AS (
24 SELECT CURRENT_DATE-6
25 UNION ALL SELECT day+1 FROM days WHERE day<CURRENT_DATE
26) SELECT d.day,COUNT(e.emergency_id) AS incident_count FROM days d
27LEFT JOIN emergencies e ON e.reported_at::DATE=d.day
28GROUP BY d.day ORDER BY d.day;
29
30-- Grouping and HAVING.
31SELECT e.emergency_id,COUNT(a.assignment_id) AS assignments FROM emergencies e
32LEFT JOIN assignments a USING(emergency_id) GROUP BY e.emergency_id
33HAVING COUNT(a.assignment_id)>=1;
