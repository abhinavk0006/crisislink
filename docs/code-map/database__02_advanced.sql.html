45CREATE VIEW v_assignment_summary AS
46SELECT a.assignment_id,a.emergency_id,r.resource_name,p.employee_id,
47 a.status,a.assigned_at,a.responder_id,p.user_id

55CREATE FUNCTION record_emergency_status() RETURNS TRIGGER
56LANGUAGE plpgsql SECURITY DEFINER SET search_path=public,pg_temp AS $$
57BEGIN
58 IF TG_OP='INSERT' OR OLD.status IS DISTINCT FROM NEW.status THEN
59 INSERT INTO emergency_status_history(emergency_id,status,remarks,changed_by,changed_at)
60 VALUES(NEW.emergency_id,NEW.status,'Automatic status audit',app_actor(),CURRENT_TIMESTAMP);
61 END IF;
62 RETURN NEW;
63END;
64$$;
65CREATE TRIGGER trg_emergency_status AFTER INSERT OR UPDATE OF status ON emergencies
66FOR EACH ROW EXECUTE FUNCTION record_emergency_status();

68CREATE FUNCTION dispatch_response(p_emergency INTEGER,p_resource INTEGER,p_responder INTEGER)
69RETURNS INTEGER LANGUAGE plpgsql SECURITY DEFINER SET search_path=public,pg_temp AS $$
70DECLARE new_id INTEGER; e emergencies; r resources; p responders;
71BEGIN
72 SELECT * INTO e FROM emergencies WHERE emergency_id=p_emergency FOR UPDATE;
73 IF NOT FOUND OR e.status='resolved' THEN RAISE EXCEPTION 'Incident is missing or resolved' USING ERRCODE='P0001'; END IF;
74 SELECT * INTO r FROM resources WHERE resource_id=p_resource FOR UPDATE;
75 IF NOT FOUND OR r.status<>'available' THEN RAISE EXCEPTION 'Resource is unavailable' USING ERRCODE='P0001'; END IF;
76 SELECT * INTO p FROM responders WHERE responder_id=p_responder FOR UPDATE;
77 IF NOT FOUND OR p.status<>'available' THEN RAISE EXCEPTION 'Responder is unavailable' USING ERRCODE='P0001'; END IF;
78 IF EXISTS(SELECT 1 FROM emergency_required_capabilities c
79 WHERE c.emergency_id=p_emergency AND NOT EXISTS(
80 SELECT 1 FROM resource_capabilities rc WHERE rc.resource_id=p_resource
81 AND rc.capability_id=c.capability_id)) THEN
82 RAISE EXCEPTION 'Resource does not cover required capabilities' USING ERRCODE='P0001';
83 END IF;
84 INSERT INTO assignments(emergency_id,resource_id,responder_id,assigned_by,assigned_at,dispatch_time,status,remarks)
85 VALUES(p_emergency,p_resource,p_responder,app_actor(),CURRENT_TIMESTAMP,CURRENT_TIMESTAMP,'dispatched','Dispatched through application')
86 RETURNING assignment_id INTO new_id;
87 UPDATE resources SET status='assigned' WHERE resource_id=p_resource;
88 UPDATE responders SET status='on_duty' WHERE responder_id=p_responder;
89 UPDATE emergencies SET status='responding',updated_at=CURRENT_TIMESTAMP WHERE emergency_id=p_emergency;
90 RETURN new_id;
91END;
92$$;

94CREATE FUNCTION finish_response(p_assignment INTEGER,p_outcome TEXT)
95RETURNS INTEGER LANGUAGE plpgsql SECURITY DEFINER SET search_path=public,pg_temp AS $$
96DECLARE a assignments; new_report INTEGER;
97BEGIN
98 SELECT * INTO a FROM assignments WHERE assignment_id=p_assignment FOR UPDATE;
99 IF NOT FOUND OR a.status NOT IN ('active','dispatched') THEN
100 RAISE EXCEPTION 'Assignment is not active' USING ERRCODE='P0001';
101 END IF;
102 IF current_setting('role')<>'crisis_coordinator' AND NOT EXISTS(
103 SELECT 1 FROM responders WHERE responder_id=a.responder_id AND user_id=app_actor()) THEN
104 RAISE EXCEPTION 'Assignment belongs to another responder' USING ERRCODE='42501';
105 END IF;
106 INSERT INTO responder_reports(emergency_id,responder_id,assignment_id,initial_assessment,actions_taken,resources_used,people_affected,outcome,reported_at)
107 VALUES(a.emergency_id,a.responder_id,a.assignment_id,'Field response','Response completed',a.resource_id::TEXT,0,p_outcome,CURRENT_TIMESTAMP)
108 RETURNING report_id INTO new_report;
109 UPDATE assignments SET status='completed' WHERE assignment_id=p_assignment;
110 UPDATE resources SET status='available' WHERE resource_id=a.resource_id;
111 UPDATE responders SET status='available' WHERE responder_id=a.responder_id;
112 IF NOT EXISTS(SELECT 1 FROM assignments WHERE emergency_id=a.emergency_id AND status IN ('active','dispatched')) THEN
113 UPDATE emergencies SET status='resolved',updated_at=CURRENT_TIMESTAMP WHERE emergency_id=a.emergency_id;
114 END IF;
115 RETURN new_report;
116END;
117$$;
