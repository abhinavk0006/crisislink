13export async function login(email: string, password: string) {
14 const db = await getDb();
15 const key = tokenHash(email.toLowerCase());
16 const blocked = await db.query<{ blocked: boolean }>(
17 "SELECT blocked_until>CURRENT_TIMESTAMP AS blocked FROM app_login_attempts WHERE attempt_key=$1",
18 [key],
19 );
20 if (blocked.rows[0]?.blocked)
21 throw new AppError(429, "Too many attempts. Try again in 15 minutes.");
22 const result = await db.query<Actor & { password_hash: string }>(
23 "SELECT u.user_id,u.full_name,u.email,u.password_hash,r.role_name FROM users u JOIN roles r USING(role_id) WHERE LOWER(u.email)=$1 AND u.status='active'",
24 [email.toLowerCase()],
25 );
26 const user = result.rows[0];
27 // Dummy hash comparison avoids an immediate response for unknown accounts.
28 const valid = verifyPassword(password, user?.password_hash || dummyHash);
29 if (!user || !valid) {
30 await db.query(
31 "INSERT INTO app_login_attempts(attempt_key,failures) VALUES($1,1) ON CONFLICT(attempt_key) DO UPDATE SET failures=app_login_attempts.failures+1,blocked_until=CASE WHEN app_login_attempts.failures>=4 THEN CURRENT_TIMESTAMP+INTERVAL '15 minutes' ELSE NULL END",
32 [key],
33 );
34 throw new AppError(401, "Invalid email or password");
35 }
36 await db.query("DELETE FROM app_login_attempts WHERE attempt_key=$1", [key]);
37 const token = randomBytes(32).toString("hex");
38 await db.query(
39 "INSERT INTO app_sessions VALUES($1,$2,CURRENT_TIMESTAMP+INTERVAL '8 hours')",
40 [tokenHash(token), user.user_id],
41 );
42 const { password_hash: _, ...actor } = user;
43 return { token, actor };
44}
45export async function authenticate(token?: string): Promise<Actor> {
46 const db = await getDb();
47 if (!token) throw new AppError(401, "Sign in to continue");
48 const result = await db.query<Actor>(
49 "SELECT u.user_id,u.full_name,u.email,r.role_name FROM app_sessions s JOIN users u USING(user_id) JOIN roles r USING(role_id) WHERE s.token_hash=$1 AND s.expires_at>CURRENT_TIMESTAMP AND u.status='active'",
50 [tokenHash(token)],
51 );
52 if (!result.rows[0]) throw new AppError(401, "Session expired");
53 return result.rows[0];
54}
