11CREATE TABLE users (

33CREATE TABLE locations (
34 location_id INTEGER GENERATED BY DEFAULT AS IDENTITY,
35 location_name VARCHAR(255),
36 address TEXT,
37 latitude DECIMAL(10,7),
38 longitude DECIMAL(10,7),
39 risk_level VARCHAR(255),
40 CONSTRAINT pk_locations PRIMARY KEY (location_id),

45CREATE TABLE emergencies (
46 emergency_id INTEGER GENERATED BY DEFAULT AS IDENTITY,

60 CONSTRAINT fk_emergency_location FOREIGN KEY (location_id)
