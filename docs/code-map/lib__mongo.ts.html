91export async function listNotes(actor: Actor, incident: number, search = "") {
92 await ensureIncident(actor, incident);
93 const col = await mongo();
94 const filter: Document = { emergency_id: incident };
95 if (search.trim()) filter.$text = { $search: search.slice(0, 100) };
96 const notes = await col
97 .find(filter)
98 .sort({ created_at: -1 })
99 .limit(100)
100 .toArray();
101 const summary = await col
102 .aggregate([
103 { $match: { emergency_id: incident } },
104 { $unwind: "$tags" },
105 { $group: { _id: "$tags", count: { $sum: 1 } } },
106 { $sort: { count: -1 } },
107 { $limit: 10 },
108 ])
109 .toArray();
110 return { notes, summary };
111}
112export async function createNote(actor: Actor, input: unknown) {
113 const d = noteInput.parse(input);
114 await ensureIncident(actor, d.emergency_id, true);

126async function ownedNote(actor: Actor, id: string) {
127 if (!ObjectId.isValid(id)) throw new AppError(400, "Invalid note ID");
128 const col = await mongo();
129 const note = await col.findOne({ _id: new ObjectId(id) });
130 if (!note) throw new AppError(404, "Note not found");
131 await ensureIncident(actor, note.emergency_id, true);
132 if (actor.role_name !== "Coordinator" && note.author_id !== actor.user_id)
133 throw new AppError(
134 403,
135 "Only the author or coordinator can change this note",
136 );
137 return { col, note };
138}
139export async function updateNote(actor: Actor, id: string, input: unknown) {
140 const { col, note } = await ownedNote(actor, id);
141 const d = noteInput.omit({ emergency_id: true }).parse(input);
