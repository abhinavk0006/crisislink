1// mongosh demonstration. Use a new database for createCollection.
2// SQL IDs 101 and 1 refer to the supplied fictional classroom seed.
3db = db.getSiblingDB('crisislink');
4db.createCollection('field_notes', {
5 validator: { $jsonSchema: {
6 bsonType: 'object',
7 required: ['emergency_id','author_id','author_name','text','tags','assessment','created_at','updated_at'],
8 properties: {
9 emergency_id: {bsonType:'int',minimum:1},
10 author_id: {bsonType:'int',minimum:1},
11 author_name: {bsonType:'string'},
12 text: {bsonType:'string',minLength:5,maxLength:2000},
13 tags: {bsonType:'array',maxItems:8,items:{bsonType:'string'}},
14 assessment: {bsonType:'object'},
15 created_at: {bsonType:'date'},
16 updated_at: {bsonType:'date'}
17 }
18 }}, validationAction:'error'
19});
20db.field_notes.createIndex({emergency_id:1,created_at:-1});
21db.field_notes.createIndex({text:'text',tags:'text'});
22
23// CREATE
24const noteId = db.field_notes.insertOne({
25 emergency_id:101,author_id:1,author_name:'Demo Coordinator',
26 text:'Training evacuation route checked and clear',tags:['safety','training'],
27 assessment:{safety:'Route clear'},created_at:new Date(),updated_at:new Date()
28}).insertedId;
29// READ and full-text feature
30db.field_notes.find({emergency_id:101}).sort({created_at:-1});
31db.field_notes.find({emergency_id:101,$text:{$search:'evacuation'}});
32// UPDATE
33db.field_notes.updateOne({_id:noteId},{$set:{text:'Training evacuation completed safely',updated_at:new Date()}});
34// Aggregation: note count by tag for an incident
35db.field_notes.aggregate([
36 {$match:{emergency_id:101}},{$unwind:'$tags'},
37 {$group:{_id:'$tags',count:{$sum:1}}},{$sort:{count:-1}}
38]);
39// DELETE only the disposable note created above
40db.field_notes.deleteOne({_id:noteId});
