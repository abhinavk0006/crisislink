1import { NextRequest, NextResponse } from "next/server";
2import { ZodError, z } from "zod";
3import { authenticate, login, logout, AppError } from "@/lib/auth";
4import {
5 dashboard,
6 createIncident,
7 dispatch,
8 complete,
9 maintenance,
10 runQuery,
11 addResourceTag,
12 updateResourceTag,
13 removeResourceTag,
14} from "@/lib/service";
15import { listNotes, createNote, updateNote, deleteNote } from "@/lib/mongo";
16import { searchIncidents } from "@/lib/semantic";
17export const runtime = "nodejs";
18export const dynamic = "force-dynamic";
19async function handler(req: NextRequest) {
20 try {
21 const path = req.nextUrl.pathname.slice("/api/".length);
22 const method = req.method;
23 if (method !== "GET") {
24 const origin = req.headers.get("origin");
25 if (origin && new URL(origin).host !== req.headers.get("host"))
26 throw new AppError(403, "Cross-origin write rejected");
27 if (Number(req.headers.get("content-length") || 0) > 16384)
28 throw new AppError(413, "Request too large");
29 }
30 const respond = (data: unknown, status = 200) =>
31 NextResponse.json(data, {
32 status,
33 headers: {
34 "Cache-Control": "no-store",
35 "X-Content-Type-Options": "nosniff",
36 },
37 });
38 if (path === "auth/login" && method === "POST") {
39 const d = z
40 .object({
41 email: z.email().max(254),
42 password: z.string().min(1).max(200),
43 })
44 .parse(await req.json());
45 const { token, actor } = await login(d.email, d.password);
46 const res = respond(actor);
47 res.cookies.set("crisis_session", token, {
48 httpOnly: true,
49 sameSite: "strict",
50 secure: req.nextUrl.protocol === "https:",
51 maxAge: 28800,
52 path: "/",
53 });
54 return res;
55 }
56 const token = req.cookies.get("crisis_session")?.value;
57 const actor = await authenticate(token);
58 if (path === "auth/me" && method === "GET") return respond(actor);
59 if (path === "auth/logout" && method === "POST") {
60 await logout(token!);
61 const res = respond({ ok: true });
62 res.cookies.delete("crisis_session");
63 return res;
64 }
65 if (path === "dashboard" && method === "GET")
66 return respond(await dashboard(actor));
67 if (path === "incidents/search" && method === "GET")
68 return respond(await searchIncidents(actor, {
69 query: req.nextUrl.searchParams.get("q"),
70 limit: req.nextUrl.searchParams.get("limit") || 5,
71 }));
72 if (path === "incidents" && method === "POST")
73 return respond(await createIncident(actor, await req.json()), 201);
74 if (path === "dispatch" && method === "POST")
75 return respond(await dispatch(actor, await req.json()), 201);
76 if (path === "complete" && method === "POST")
77 return respond(await complete(actor, await req.json()));
78 if (path === "maintenance" && method === "POST")
79 return respond(await maintenance(actor, await req.json()), 201);
80 if (path === "resource-tags") {
81 if (method === "POST")
82 return respond(await addResourceTag(actor, await req.json()), 201);
83 if (method === "PATCH")
84 return respond(await updateResourceTag(actor, await req.json()));
85 if (method === "DELETE")
86 return respond(await removeResourceTag(actor, await req.json()));
87 }
88 if (path === "queries" && method === "GET")
89 return respond(
90 await runQuery(actor, req.nextUrl.searchParams.get("key") || "join"),
91 );
92 if (path === "notes" && method === "GET") {
93 const id = z.coerce
94 .number()
95 .int()
96 .positive()
97 .parse(req.nextUrl.searchParams.get("incident"));
98 return respond(
99 await listNotes(
100 actor,
101 id,
102 req.nextUrl.searchParams.get("search") || "",
103 ),
104 );
105 }
106 if (path === "notes" && method === "POST")
107 return respond(await createNote(actor, await req.json()), 201);
108 if (path.startsWith("notes/")) {
109 const id = path.slice(6);
110 if (method === "PATCH")
111 return respond(await updateNote(actor, id, await req.json()));
112 if (method === "DELETE") return respond(await deleteNote(actor, id));
113 }
114 throw new AppError(404, "Endpoint not found");
115 } catch (e) {
116 let status = 500,
117 message = "Request could not be completed";
118 if (e instanceof AppError) {
119 status = e.status;
120 message = e.message;
121 } else if (e instanceof ZodError) {
122 status = 400;
123 message = e.issues
124 .map((i) => `${i.path.join(".")}: ${i.message}`)
125 .join("; ");
126 } else if (e instanceof SyntaxError) {
127 status = 400;
128 message = "Invalid JSON";
129 } else if (
130 e instanceof Error &&
131 /Mongo(Network|ServerSelection|TopologyClosed|NotConnected)/.test(e.name)
132 ) {
133 status = 503;
134 message =
135 "MongoDB is unavailable. The operation was not confirmed; retry after reconnecting.";
136 } else {
137 const err = e as { code?: string; message?: string };
138 if (err.code === "42501") {
139 status = 403;
140 message = "Database permission denied";
141 } else if (
142 ["23503", "23505", "23514", "23502", "P0001"].includes(err.code || "")
143 ) {
144 status = 409;
145 message =
146 err.code === "P0001"
147 ? err.message!
148 : "The change conflicts with database constraints";
149 } else console.error("API failure", e);
150 }
151 return NextResponse.json(
152 { error: message },
153 { status, headers: { "Cache-Control": "no-store" } },
154 );
155 }
156}
157export { handler as GET, handler as POST, handler as PATCH, handler as DELETE };
