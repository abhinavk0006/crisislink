1-- Additive, idempotent migration: preserves the 17-table operational model.
2CREATE EXTENSION IF NOT EXISTS vector;
3ALTER TABLE emergencies ADD COLUMN IF NOT EXISTS embedding vector(384);
4ALTER TABLE emergencies ADD COLUMN IF NOT EXISTS embedding_model VARCHAR(160);
5ALTER TABLE emergencies ADD COLUMN IF NOT EXISTS embedded_at TIMESTAMPTZ;
6
7CREATE INDEX IF NOT EXISTS idx_emergency_embedding_hnsw
8ON emergencies USING hnsw (embedding vector_cosine_ops)
9WHERE embedding IS NOT NULL;
10
11-- A description edit must never leave a searchable vector for the old text.
12CREATE OR REPLACE FUNCTION invalidate_incident_embedding() RETURNS TRIGGER
13LANGUAGE plpgsql SET search_path=public,pg_temp AS $$
14BEGIN
15 IF OLD.description IS DISTINCT FROM NEW.description THEN
16 NEW.embedding := NULL;
17 NEW.embedding_model := NULL;
18 NEW.embedded_at := NULL;
19 END IF;
20 RETURN NEW;
21END;
22$$;
23REVOKE EXECUTE ON FUNCTION invalidate_incident_embedding() FROM PUBLIC;
24DROP TRIGGER IF EXISTS trg_invalidate_incident_embedding ON emergencies;
25CREATE TRIGGER trg_invalidate_incident_embedding
26BEFORE UPDATE OF description ON emergencies
27FOR EACH ROW EXECUTE FUNCTION invalidate_incident_embedding();
