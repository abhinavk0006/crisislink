10export const DUPLICATE_WINDOW_HOURS = 24;
11// Candidate-screening heuristic, not a probability or proof of identity.
12export const MAX_DUPLICATE_DISTANCE = 0.65;
13const shared = globalThis as unknown as { crisisReviewSecret?: Buffer };
14const secret = shared.crisisReviewSecret ||= randomBytes(32);
15type Draft = z.infer<typeof incidentInput>;
16type Review = { id:string;user:number;draft:string;expires:number;candidates:{id:number;fingerprint:string}[] };
17const hash = (value: unknown) => createHash("sha256").update(JSON.stringify(value)).digest("hex");
18const fingerprint = (r: any) => hash([r.description,r.location_id,r.reported_at]);
19function sign(review: Review) {
20 const payload = Buffer.from(JSON.stringify(review)).toString("base64url");
21 return payload + "." + createHmac("sha256",secret).update(payload).digest("base64url");
22}
23function verify(token: string, actor: Actor, draft: Draft): Review {
24 try {
25 const [payload,signature,...extra] = token.split(".");
26 if (extra.length) throw Error();
27 const expected = createHmac("sha256",secret).update(payload).digest();
28 const actual = Buffer.from(signature,"base64url");
29 if (actual.length !== expected.length || !timingSafeEqual(actual,expected)) throw Error();
30 const review = JSON.parse(Buffer.from(payload,"base64url").toString()) as Review;
31 if (review.user !== actor.user_id || review.draft !== hash(draft) || review.expires < Date.now()) throw Error();
32 return review;
33 } catch { throw new AppError(409,"The report changed or its review expired. Check for existing incidents again."); }
34}

36export const DUPLICATE_SQL = `WITH nearest AS MATERIALIZED (
37 SELECT e.*, e.embedding <=> $1::vector AS cosine_distance
38 FROM emergencies e
39 WHERE e.location_id=$2 AND e.status IN ('reported','responding')
40 AND e.reported_at>=CURRENT_TIMESTAMP-INTERVAL '24 hours'
41 AND e.reported_at<=CURRENT_TIMESTAMP
42 AND e.embedding IS NOT NULL AND e.embedding_model=$3
43 ORDER BY e.embedding <=> $1::vector LIMIT 5
44)
45SELECT e.emergency_id,e.description,e.location_id,
46 e.reported_at::text AS reported_at,e.status,t.type_name,
47 e.cosine_distance,
48 (SELECT COALESCE(json_agg(a),'[]'::json) FROM v_assignment_summary a
49 WHERE a.emergency_id=e.emergency_id AND a.status IN ('active','dispatched')) AS active_assignments
50FROM nearest e JOIN emergency_types t USING(type_id)
51ORDER BY e.cosine_distance`;

53export async function reviewIncident(actor: Actor, input: unknown) {
54 requireRole(actor,"Coordinator");
55 const draft = incidentInput.parse(input);
56 // No report is written by this read-only review. Index writes are derived metadata only.
57 await indexIncidents();
58 const vector = vectorLiteral(await embedText(draft.description));
59 const candidates = await asActor(actor, async (tx) => {
60 await tx.exec("SET LOCAL hnsw.iterative_scan = 'strict_order'");
61 const location = (await tx.query("SELECT 1 FROM locations WHERE location_id=$1",[draft.location_id])).rows[0];
62 if (!location) throw new AppError(400,"Choose an existing location");
63 const rows = (await tx.query<any>(DUPLICATE_SQL,[vector,draft.location_id,EMBEDDING_ID])).rows;
64 return rows.filter(r => r.cosine_distance <= MAX_DUPLICATE_DISTANCE);
65 });
66 return { candidates, windowHours: DUPLICATE_WINDOW_HOURS, maxDistance: MAX_DUPLICATE_DISTANCE,
67 reviewToken: sign({ id:randomUUID(),user:actor.user_id,draft:hash(draft),expires:Date.now()+10*60*1000,
68 candidates:candidates.map(r=>({id:r.emergency_id,fingerprint:fingerprint(r)})) }) };
69}

71export async function confirmIncidentReport(actor: Actor, input: unknown) {
72 requireRole(actor,"Coordinator");
73 const d = z.object({ draft:incidentInput, action:z.enum(["create","attach"]),
74 reviewToken:z.string().max(10000).optional(), emergency_id:z.number().int().positive().optional(),
75 bypass_review:z.literal(true).optional(), request_id:z.uuid().optional(),
76 }).parse(input);
77 // Explicit emergency-reporting fallback: never silently claim a model check succeeded.
78 if (!d.reviewToken) {
79 if (d.action!=="create" || !d.bypass_review || !d.request_id)
80 throw new AppError(400,"Review the report or explicitly create it without a duplicate check.");
81 const row = await createIncident(actor,d.draft,d.request_id);
82 return { ...row, action:"created", reviewed:false };
83 }
84 const review = verify(d.reviewToken,actor,d.draft);
85 if (d.action==="create") {
86 const row = await createIncident(actor,d.draft,review.id);
87 return { ...row, action:"created", reviewed:true };
88 }
89 const candidate = review.candidates.find(r=>r.id===d.emergency_id);
90 if (!candidate) throw new AppError(400,"Choose an incident from this report's review results.");
91 const incident = d.emergency_id!;
92 const col = await mongo();
93 const existing = await col.findOne({report_review_id:review.id,author_id:actor.user_id,emergency_id:incident});
94 if (existing) return {action:"attached",emergency_id:incident,note_id:existing._id.toHexString()};
95 await asActor(actor,async(tx)=>{
96 const current = (await tx.query<any>(
97 "SELECT description,location_id,reported_at::text AS reported_at FROM emergencies WHERE emergency_id=$1 AND status IN ('reported','responding') AND reported_at>=CURRENT_TIMESTAMP-INTERVAL '24 hours' AND reported_at<=CURRENT_TIMESTAMP FOR UPDATE",
98 [incident],
99 )).rows[0];
100 if (!current || fingerprint(current)!==candidate.fingerprint)
101 throw new AppError(409,"That incident changed or is no longer eligible. Review the report again.");
102 await tx.query("INSERT INTO app_incident_report_receipts(review_id,user_id,action,emergency_id,draft_hash) VALUES($1,$2,'attach',$3,$4) ON CONFLICT(review_id) DO NOTHING",[review.id,actor.user_id,incident,hash(d.draft)]);
103 const receipt=(await tx.query<any>("SELECT * FROM app_incident_report_receipts WHERE review_id=$1",[review.id])).rows[0];
104 if(receipt.user_id!==actor.user_id || receipt.action!=="attach" || receipt.emergency_id!==incident)
105 throw new AppError(409,"This report already has a different decision. Review it again.");
106 });
107 const saved = await saveAdditionalReport(actor,incident,review.id,d.draft);
108 return {action:"attached",emergency_id:incident,note_id:saved.id};
109}
